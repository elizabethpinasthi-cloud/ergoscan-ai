/* =====================================================
   INITIALIZE MODEL
===================================================== */

async function loadModel() {

  setConnection(
    "⏳ Loading AI model..."
  );

  try {

    await tf.setBackend("webgl");

    await tf.ready();

    const model =
      poseDetection.SupportedModels.MoveNet;

    detector =
      await poseDetection.createDetector(
        model,
        {
          modelType:
            poseDetection.movenet.modelType.SINGLEPOSE_LIGHTNING,

          enableSmoothing:
            true
        }
      );

    modelReady = true;

    setConnection(
      "● AI Ready"
    );

  }

  catch(error) {

    console.error(error);

    setConnection(
      "⚠ AI gagal dimuat"
    );

    alert(
      "Model AI gagal dimuat. Periksa koneksi internet."
    );

  }

}


/* =====================================================
   START CAMERA
===================================================== */

async function startCamera() {

  try {

    if (!modelReady) {

      await loadModel();

    }


    if (!navigator.mediaDevices ||
        !navigator.mediaDevices.getUserMedia) {

      throw new Error(
        "Browser tidak mendukung kamera."
      );

    }


    stream =
      await navigator.mediaDevices.getUserMedia({

        video: {

          facingMode:
            "environment",

          width: {
            ideal: 1280
          },

          height: {
            ideal: 720
          }

        },

        audio: false

      });


    video.srcObject =
      stream;


    await video.play();


    document.getElementById(
      "cameraStatus"
    ).textContent =
      "● Camera Active";


    document.getElementById(
      "cameraStatus"
    ).style.color =
      "#6ee7b7";


    resizeCanvas();


    detectLoop();


  }

  catch(error) {

    console.error(error);

    let message =
      "Kamera tidak dapat digunakan.";

    if (
      error.name ===
      "NotAllowedError"
    ) {

      message +=
        "\n\nIzin kamera ditolak.";

    }

    if (
      error.name ===
      "NotFoundError"
    ) {

      message +=
        "\n\nKamera tidak ditemukan.";

    }

    if (
      error.name ===
      "NotReadableError"
    ) {

      message +=
        "\n\nKamera sedang digunakan aplikasi lain.";

    }


    alert(message);

  }

}

}

/* =====================================================
   DETECTION LOOP
===================================================== */

async function detectLoop() {

  if (!stream ||
      !detector) {

    return;

  }


  try {

    const poses =
      await detector.estimatePoses(
        video,
        {
          flipHorizontal:
            true
        }
      );


    ctx.clearRect(
      0,
      0,
      canvas.width,
      canvas.height
    );


    if (
      poses.length > 0
    ) {

      const pose =
        poses[0];

      drawPose(
        pose
      );


      analyzePose(
        pose
      );

    }

  }

  catch(error) {

    console.error(
      "Detection error:",
      error
    );

  }


  animationId =
    requestAnimationFrame(
      detectLoop
    );

}


/* =====================================================
   DRAW SKELETON
===================================================== */

function drawPose(
  pose
) {

  const keypoints =
    pose.keypoints;


  const pairs = [

    ["left_shoulder","right_shoulder"],

    ["left_shoulder","left_elbow"],

    ["left_elbow","left_wrist"],

    ["right_shoulder","right_elbow"],

    ["right_elbow","right_wrist"],

    ["left_shoulder","left_hip"],

    ["right_shoulder","right_hip"],

    ["left_hip","right_hip"],

    ["left_hip","left_knee"],

    ["left_knee","left_ankle"],

    ["right_hip","right_knee"],

    ["right_knee","right_ankle"],

    ["left_eye","right_eye"]

  ];


  const kp = {};


  keypoints.forEach(
    point => {

      kp[point.name] =
        point;

    }
  );


  ctx.lineWidth =
    4;

  ctx.strokeStyle =
    "#34d399";


  pairs.forEach(
    pair => {

      const a =
        kp[pair[0]];

      const b =
        kp[pair[1]];


      if (
        a &&
        b &&
        a.score > .35 &&
        b.score > .35
      ) {

        ctx.beginPath();

        ctx.moveTo(
          a.x,
          a.y
        );

        ctx.lineTo(
          b.x,
          b.y
        );

        ctx.stroke();

      }

    }
  );


  keypoints.forEach(
    point => {

      if (
        point.score > .35
      ) {

        ctx.beginPath();

        ctx.arc(
          point.x,
          point.y,
          6,
          0,
          Math.PI * 2
        );

        ctx.fillStyle =
          "#ffffff";

        ctx.fill();

      }

    }
  );

}


/* =====================================================
   ANGLE FUNCTION
===================================================== */

function angle(
  a,
  b,
  c
) {

  const ab = {

    x: a.x - b.x,

    y: a.y - b.y

  };


  const cb = {

    x: c.x - b.x,

    y: c.y - b.y

  };


  const dot =
    ab.x * cb.x +
    ab.y * cb.y;


  const magAB =
    Math.sqrt(
      ab.x ** 2 +
      ab.y ** 2
    );


  const magCB =
    Math.sqrt(
      cb.x ** 2 +
      cb.y ** 2
    );


  if (
    magAB === 0 ||
    magCB === 0
  ) {

    return 0;

  }


  let cosine =
    dot /
    (magAB * magCB);


  cosine =
    Math.max(
      -1,
      Math.min(
        1,
        cosine
      )
    );


  return (
    Math.acos(cosine) *
    180 /
    Math.PI
  );

}


/* =====================================================
   ANALYZE POSE
===================================================== */

function analyzePose(
  pose
) {

  const kp = {};

  pose.keypoints.forEach(
    p => {

      kp[p.name] =
        p;

    }
  );


  const required = [

    "left_shoulder",
    "right_shoulder",
    "left_elbow",
    "right_elbow",
    "left_wrist",
    "right_wrist",
    "left_hip",
    "right_hip",
    "left_knee",
    "right_knee",
    "left_ankle",
    "right_ankle",
    "nose"

  ];


  for (
    const name of required
  ) {

    if (
      !kp[name] ||
      kp[name].score < .35
    ) {

      return;

    }

  }


  /* --------------------------------------------------
     SHOULDER CENTER
  -------------------------------------------------- */

  const shoulder = {

    x:
      (
        kp.left_shoulder.x +
        kp.right_shoulder.x
      ) / 2,

    y:
      (
        kp.left_shoulder.y +
        kp.right_shoulder.y
      ) / 2

  };


  /* --------------------------------------------------
     HIP CENTER
  -------------------------------------------------- */

  const hip = {

    x:
      (
        kp.left_hip.x +
        kp.right_hip.x
      ) / 2,

    y:
      (
        kp.left_hip.y +
        kp.right_hip.y
      ) / 2

  };


  /* --------------------------------------------------
     KNEE CENTER
  -------------------------------------------------- */

  const knee = {

    x:
      (
        kp.left_knee.x +
        kp.right_knee.x
      ) / 2,

    y:
      (
        kp.left_knee.y +
        kp.right_knee.y
      ) / 2

  };


  /* --------------------------------------------------
     NECK
  -------------------------------------------------- */

  const neckAngle =
    calculateVerticalAngle(
      kp.nose,
      shoulder
    );


  /* --------------------------------------------------
     TRUNK
  -------------------------------------------------- */

  const trunkAngle =
    calculateVerticalAngle(
      shoulder,
      hip
    );


  /* --------------------------------------------------
     UPPER ARM
  -------------------------------------------------- */

  const leftUpper =
    angle(
      kp.left_hip,
      kp.left_shoulder,
      kp.left_elbow
    );


  const rightUpper =
    angle(
      kp.right_hip,
      kp.right_shoulder,
      kp.right_elbow
    );


  const upperArm =
    Math.max(
      leftUpper,
      rightUpper
    );


  /* --------------------------------------------------
     LOWER ARM
  -------------------------------------------------- */

  const leftLower =
    angle(
      kp.left_shoulder,
      kp.left_elbow,
      kp.left_wrist
    );


  const rightLower =
    angle(
      kp.right_shoulder,
      kp.right_elbow,
      kp.right_wrist
    );


  const lowerArm =
    Math.min(
      leftLower,
      rightLower
    );


  /* --------------------------------------------------
     WRIST
  -------------------------------------------------- */

  const wrist =
    calculateWristAngle(
      kp.left_elbow,
      kp.left_wrist,
      kp.left_index || kp.left_wrist
    );


  /* --------------------------------------------------
     CONFIDENCE
  -------------------------------------------------- */

  const confidence =
    pose.keypoints.reduce(
      (
        total,
        point
      ) =>
        total +
        point.score,
      0
    ) /
    pose.keypoints.length;


  const rula =
    calculateRULA({

      neck:
        neckAngle,

      trunk:
        trunkAngle,

      upperArm:
        upperArm,

      lowerArm:
        lowerArm,

      wrist:
        wrist,

      load:
        Number(
          document.getElementById(
            "load"
          ).value
        ) || 0

    });


  const reba =
    calculateREBA({

      neck:
        neckAngle,

      trunk:
        trunkAngle,

      upperArm:
        upperArm,

      lowerArm:
        lowerArm,

      wrist:
        wrist,

      load:
        Number(
          document.getElementById(
            "load"
          ).value
        ) || 0

    });


  currentResult = {

    rula:
      rula.score,

    rulaLevel:
      rula.level,

    reba:
      reba.score,

    rebaLevel:
      reba.level,

    confidence:
      confidence,

    angles: {

      neck:
        neckAngle,

      trunk:
        trunkAngle,

      upperArm:
        upperArm,

      lowerArm:
        lowerArm,

      wrist:
        wrist

    }

  };


  updateUI(
    currentResult
  );

}


/* =====================================================
   VERTICAL ANGLE
===================================================== */

function calculateVerticalAngle(
  top,
  bottom
) {

  const dx =
    top.x -
    bottom.x;

  const dy =
    bottom.y -
    top.y;


  return Math.abs(
    Math.atan2(
      dx,
      dy
    ) *
    180 /
    Math.PI
  );

}


/* =====================================================
   WRIST ANGLE
===================================================== */

function calculateWristAngle(
  elbow,
  wrist,
  hand
) {

  return angle(
    elbow,
    wrist,
    hand
  );

}


/* =====================================================
   RULA
===================================================== */

function calculateRULA(
  p
) {

  let score = 1;


  /* Neck */

  if (
    p.neck <= 10
  ) {

    score += 0;

  }

  else if (
    p.neck <= 20
  ) {

    score += 1;

  }

  else {

    score += 2;

  }


  /* Trunk */

  if (
    p.trunk <= 20
  ) {

    score += 0;

  }

  else if (
    p.trunk <= 60
  ) {

    score += 1;

  }

  else {

    score += 2;

  }


  /* Upper arm */

  if (
    p.upperArm > 90
  ) {

    score += 3;

  }

  else if (
    p.upperArm > 45
  ) {

    score += 2;

  }

  else if (
    p.upperArm > 20
  ) {

    score += 1;

  }


  /* Lower arm */

  if (
    p.lowerArm < 60 ||
    p.lowerArm > 100
  ) {

    score += 1;

  }


  /* Wrist */

  if (
    p.wrist > 15
  ) {

    score += 1;

  }


  /* Force */

  if (
    p.load > 10
  ) {

    score += 2;

  }

  else if (
    p.load >= 2
  ) {

    score += 1;

  }


  score =
    Math.min(
      7,
      Math.max(
        1,
        score
      )
    );


  let level;


  if (
    score <= 2
  ) {

    level =
      "Low";

  }

  else if (
    score <= 4
  ) {

    level =
      "Medium";

  }

  else if (
    score <= 6
  ) {

    level =
      "High";

  }

  else {

    level =
      "Very High";

  }


  return {

    score,
    level

  };

}


/* =====================================================

    p.neck > 20

  }

  else if (
    p.neck > 10
  ) {

    score += 1;

  }


  /* Upper arm */

    p.upperArm > 90

  else if (
    p.upperArm > 45
  ) {

  ) {

    score += 1;

  }


  /* Lower arm */

  if (
    p.lowerArm < 60 ||
    p.lowerArm > 100
  ) {

    score += 1;

  }


  /* Wrist */

  if (
    p.wrist > 15
  ) {

    score += 1;

  }


  /* Load */

  if (
    p.load > 10
  ) {

    score += 2;

  }

  else if (
    p.load >= 5
  ) {

    score += 1;

  }


  score =

    );


  let level;


  if (
    score <= 3

  else if (

  ) {

  }

  else if (
    score <= 10
  ) {

    level =
      "High";

  }

  else {

    level =

  }


  return {

}
